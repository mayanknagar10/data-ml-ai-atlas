# Full-Chapter Platform Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the schema, generated artifacts, reader, validation, and first six verified chapters required to scale the Atlas from summaries to a full curriculum.

**Architecture:** `source/atlas.json` remains canonical. Shared Python contract helpers derive lightweight index records and validate full chapters; the build emits one small global index plus one JSON file per lesson. A tested browser loader fetches chapters on demand, while a backward-compatible reader renders both legacy summaries and structured chapters during the multi-module migration.

**Tech Stack:** Python 3 standard library, JSON, unittest, vanilla JavaScript, Node.js smoke tests, HTML/CSS/SVG, existing static GitHub Pages build.

## Global Constraints

- Do not hand-edit generated `data.js`, `curriculum.json`, `resources.json`, `CURRICULUM.md`, `book.html`, or `labs/`.
- Preserve all existing lesson slugs and local progress keys.
- `scripts/build.py` must stop rewriting `source/atlas.json`.
- Missing `editorial.status` means `summary`; it never means `verified`.
- Full chapter bodies are excluded from the initial `data.js` payload.
- `data.js` must remain at or below 2,000,000 bytes.
- Old lesson fields render until their module is migrated.
- User-authored prose, code, source notes, and visuals live only in `source/atlas.json`.
- All new browser text is escaped before insertion unless it is generated by a trusted renderer from structured data.
- Theme persistence and static GitHub Pages compatibility remain unchanged.
- Every implementation task follows a failing-test -> implementation -> passing-test cycle.

- [ ] **Step 3: Implement the contract helpers**

Create `scripts/chapter_contract.py` with this public structure:

```python
from __future__ import annotations

import re
from typing import Any

EDITORIAL_STATES = {"summary", "draft", "chapter-complete", "verified"}
FULL_CHAPTER_FIELDS = {
    "description",
    "prerequisites",
    "learningObjectives",
    "deepSections",
    "workedExamples",
    "followUps",
    "exercises",
    "sourceNotes",
    "editorial",
}
BODY_FIELDS = {
    "why", "intuition", "deepDive", "deepSections", "math", "code", "lab",
    "workedExample", "workedExamples", "commonMistakes", "followUps",
    "production", "exercises", "sourceNotes", "visual", "visuals",
}
INDEX_FIELDS = (
    "slug", "title", "module", "priority", "roles", "interviewAnswer",
    "keyPoints", "resources", "featured", "description", "prerequisites",
    "learningObjectives", "editorial",
)


def word_count(value: Any) -> int:
    if isinstance(value, str):
        return len(re.findall(r"\b[\w'-]+\b", value, flags=re.UNICODE))
    if isinstance(value, list):
        return sum(word_count(item) for item in value)
    if isinstance(value, dict):
        return sum(word_count(item) for item in value.values())
    return 0


def editorial_status(lesson: dict) -> str:
    return lesson.get("editorial", {}).get("status", "summary")


def lesson_index_record(lesson: dict) -> dict:
    record = {key: lesson[key] for key in INDEX_FIELDS if key in lesson}
    visuals = lesson.get("visuals") or ([lesson["visual"]] if lesson.get("visual") else [])
    record["status"] = editorial_status(lesson)
    record["hasLab"] = bool(lesson.get("lab"))
    record["labGoal"] = lesson.get("lab", {}).get("goal", "")
    record["visualCount"] = len(visuals)
    record["visualSummaries"] = [
        {"type": visual.get("type", ""), "title": visual.get("title", "")}
        for visual in visuals
    ]
    record["estimatedMinutes"] = max(1, round(word_count(lesson) / 220))
    return record


def chapter_errors(lesson: dict, lesson_slugs: set[str], resources: dict[str, dict]) -> list[str]:
    errors: list[str] = []
    status = editorial_status(lesson)
    if status not in EDITORIAL_STATES:
        errors.append(f"{lesson.get('slug')}: invalid editorial status {status}")
    for prerequisite in lesson.get("prerequisites", []):
        if prerequisite not in lesson_slugs:
            errors.append(f"{lesson.get('slug')}: unknown prerequisite {prerequisite}")
    if status in {"chapter-complete", "verified"}:
        missing = sorted(field for field in FULL_CHAPTER_FIELDS if not lesson.get(field))
        errors.extend(f"{lesson.get('slug')}: missing {field}" for field in missing)
        if word_count(lesson.get("description")) < 150:
            errors.append(f"{lesson.get('slug')}: description below 150 words")
        if len(lesson.get("learningObjectives", [])) < 3:
            errors.append(f"{lesson.get('slug')}: fewer than 3 learning objectives")
        if word_count(lesson.get("deepSections")) < 500:
            errors.append(f"{lesson.get('slug')}: deep sections below 500 words")
        if len(lesson.get("workedExamples", [])) < 1:
            errors.append(f"{lesson.get('slug')}: no worked example")
        if len(lesson.get("followUps", [])) < 2 or any(
            not isinstance(item, dict) or not item.get("question") or not item.get("answer")
            for item in lesson.get("followUps", [])
        ):
            errors.append(f"{lesson.get('slug')}: follow-ups require answered question objects")
        if len(lesson.get("exercises", [])) < 3 or any(
            not item.get("prompt") or not item.get("solution")
            for item in lesson.get("exercises", [])
        ):
            errors.append(f"{lesson.get('slug')}: exercises require prompts and solutions")
        for note in lesson.get("sourceNotes", []):
            if note.get("resource") not in resources:
                errors.append(f"{lesson.get('slug')}: unknown resource {note.get('resource')}")
            if word_count(note.get("contribution", "")) < 5:
                errors.append(f"{lesson.get('slug')}: source contribution is too short")
    return errors
```

- [ ] **Step 3: Implement the UMD loader**

Use this contract:

```javascript
(function (root, factory) {
  const api = factory();
  if (typeof module === "object" && module.exports) module.exports = api;
  root.AtlasChapterLoader = api;
})(typeof globalThis !== "undefined" ? globalThis : this, function () {
  function create(fetchImpl, basePath = "lesson-data") {
    const cache = new Map();
    async function load(slug, { refresh = false } = {}) {
      if (refresh) cache.delete(slug);
      if (cache.has(slug)) return cache.get(slug);
      const pending = Promise.resolve(fetchImpl(`${basePath}/${encodeURIComponent(slug)}.json`))
        .then(response => {
          if (!response.ok) throw new Error(`Lesson request failed (${response.status})`);
          return response.json();
        })
        .catch(error => {
          cache.delete(slug);
          throw error;
        });
      cache.set(slug, pending);
      return pending;
    }
    function clear(slug) {
      if (slug) cache.delete(slug);
      else cache.clear();
    }
    return { load, clear };
  }
  return { create };
});
```
